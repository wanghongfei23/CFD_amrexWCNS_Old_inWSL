11// MUSCL插值函数
12// 输入：三个相邻点的值q1, q2, q3
13// 输出：插值后的结果
14AMREX_FORCE_INLINE
15amrex::Real musclInterpolation(amrex::Real q1, amrex::Real q2, amrex::Real q3)
16{
17 using namespace amrex;
18 amrex::Real delta; // 插值增量
19 amrex::Real deltam, deltap; // 左右差分
20 deltam = q2 - q1; // 左侧差分
21 deltap = q3 - q2; // 右侧差分
22
23 // minmod限制器
24 amrex::Real beta = 1.0; // 限制器参数
25 if (deltap > 0) {
26 delta = max(0.0, max(min(beta * deltam, deltap), min(deltam, beta * deltap)));
27 } else {
28 delta = min(0.0, min(max(beta * deltam, deltap), max(deltam, beta * deltap)));
29 }
30 return q2 + delta * 0.5; // 返回插值结果
31}

33// 重构函数
34// 输入：网格坐标i,j,k，原始变量数组primin_array，变量数量NUM_STATE，问题参数pp，法向量norm
35// 输出：左右侧重构后的变量值
36// 模板参数dir：重构方向

38template <int dir = 0>
39AMREX_GPU_HOST_DEVICE
40 // AMREX_FORCE_INLINE
41 amrex::GpuArray<VarArray, 2>
42 reconstruction(int i, int j, int k,
43 amrex::Array4<amrex::Real const> const& primin_array,
44 int NUM_STATE, ProbParm const& pp, amrex::GpuArray<amrex::Real, 3> norm = { 1, 0, 0 })
45{
46 amrex::Vector<amrex::Real> res(NUM_STATE * 2); // 结果向量，存储左右侧重构值
47 // auto iter = res.begin();
48 for (int ivar = 0; ivar < NUM_STATE; ivar++) { // 遍历所有变量
49 amrex::GpuArray<amrex::Real, 4> state_temp; // 临时存储四个点的值
50 // 沿指定方向遍历，获取四个点的值
51 traverse_in_direction_with_ii<dir>(i, j, k, -2, 1, [&](int li, int lj, int lk, int ii) {
52 state_temp[ii] = primin_array(li, lj, lk, ivar);
53 });
54 // 计算左侧重构值
55 res[ivar] = musclInterpolation(state_temp[0], state_temp[1], state_temp[2]);
56 // 计算右侧重构值
57 res[ivar + NUM_STATE] = musclInterpolation(state_temp[3], state_temp[2], state_temp[1]);
58 }
59
60 return res;
61}
