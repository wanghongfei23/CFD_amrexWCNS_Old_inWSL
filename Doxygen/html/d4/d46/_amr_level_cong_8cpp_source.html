197Real AmrLevelCong::estTimeStep(Real)
198{
199 // 这只是一个初始的虚拟值
200 ProbParm const& pp = *d_prob_parm;
201 Real dt_est = 1.0e+20;
202
203 GpuArray<Real, BL_SPACEDIM> dx = geom.CellSizeArray();
204 GpuArray<Real, BL_SPACEDIM> prob_lo = geom.ProbLoArray();
205 const Real cur_time = state[State_Type].curTime();
206 const MultiFab& S_new = get_new_data(State_Type);
207 Real pred_time = cur_time;
208 if (cur_time > 0._rt) {
209 pred_time += 0.5_rt * parent->dtLevel(level);
210 }
211
212 // 初始化 tau_xyz 为零
213 for (int dir = 0; dir < BL_SPACEDIM; ++dir) {
214 tau_xyz[dir] = 0.0;
215 }
216
217#ifdef AMREX_USE_OMP
218#pragma omp parallel reduction(min : dt_est)
219#endif
220 {
221 FArrayBox umaxFAB; // 用于存储局部最大特征速度的FArrayBox
222 for (MFIter mfi(S_new, TilingIfNotGPU()); mfi.isValid(); ++mfi) {
223 const Box& bx = mfi.tilebox(); // 当前网格块的范围
224 umaxFAB.resize(bx, BL_SPACEDIM); // 每个网格点存储BL_SPACEDIM个特征速度
225 auto umaxArray = umaxFAB.array(); // 获取FArrayBox的数组
226
227 // 获取密度、动量和总能量的数组视图
228 Array4<Real const> const& state_array = S_new[mfi].const_array();
229
230 // 并行计算每个网格点的局部最大特征速度
231 ParallelFor(bx, [=] AMREX_GPU_DEVICE(int i, int j, int k) {
232 cal_u_max(i, j, k, state_array, umaxArray, pp, dx);
233 });
234
235 // 计算当前网格块的最大特征速度
236 GpuArray<Real, BL_SPACEDIM> local_tau_xyz; // 局部最大特征速度
237 for (int dir = 0; dir < BL_SPACEDIM; ++dir) {
238 local_tau_xyz[dir] = umaxFAB.norm<RunOn::Device>(0, dir, 1);
239 }
240
241 // 更新全局最大特征速度
242 for (int dir = 0; dir < BL_SPACEDIM; ++dir) {
243 tau_xyz[dir] = amrex::max(tau_xyz[dir], local_tau_xyz[dir]);
244 }
245 }
246 }
247
248 // 归约操作：确保 tau_xyz 在所有进程中一致
249 for (int dir = 0; dir < BL_SPACEDIM; ++dir) {
250 ParallelDescriptor::ReduceRealMax(tau_xyz[dir]);
251 }
252
253 // 计算时间步长估计值（放在循环外）
254 Real norminator = 0.0;
255 for (int dir = 0; dir < BL_SPACEDIM; ++dir) {
256 norminator += tau_xyz[dir];
257 }
258 if (norminator > 1.e-100) {
259 dt_est = amrex::min(dt_est, 1.0 / norminator);
260 }
261
262 // 应用 CFL 条件
263 dt_est *= cfl;
264
265 if (verbose) {
266 amrex::Print() << "AmrLevelCong::estTimeStep at level " << level
267 << ": dt_est = " << dt_est << std::endl;
268 amrex::Print() << "AmrLevelCong::estTimeStep: tau_xyz = ";
269 for (int dir = 0; dir < BL_SPACEDIM; ++dir) {
270 amrex::Print() << tau_xyz[dir] << " ";
271 }
272 amrex::Print() << std::endl;
273 }
274
275 return dt_est;
276}

297void AmrLevelCong::computeInitialDt(int finest_level,
298 int /*sub_cycle*/, // 未使用的参数
299 Vector<int>& n_cycle,
300 const Vector<IntVect>& /*ref_ratio*/, // 未使用的参数
301 Vector<Real>& dt_level,
302 Real stop_time)
303{
304 // 此函数应仅在最粗级别（level == 0）执行。
305 // 如果当前级别大于0，立即退出函数。
306 if (level > 0)
307 return;
308
309 // 初始化`dt_0`为一个大值，以找到所有级别中的最小时间步长。
310 Real dt_0 = 1.0e+100;
311
312 // `n_factor` 累积级别之间子循环的缩放因子。
313 int n_factor = 1;
314
315 // 从最粗级别到最细级别遍历所有级别。
316 for (int i = 0; i <= finest_level; i++) {
317 // 使用其特定逻辑计算当前级别的初始时间步长。
318 dt_level[i] = getLevel(i).initialTimeStep();
319
320 // 更新当前级别的子循环因子。
321 n_factor *= n_cycle[i];
322
323 // 找到所有级别中的最小缩放时间步长。
324 dt_0 = amrex::min(dt_0, n_factor * dt_level[i]);
325 }
326
327 // 调整计算的`dt_0`以确保它不超过停止时间。
328 const Real eps = 0.001 * dt_0; // 小公差以避免稍微超过stop_time。
329 Real cur_time = state[State_Type].curTime(); // 当前模拟时间。
330 if (stop_time >= 0.0) {
331 // 如果应用`dt_0`后的总时间超过stop_time，调整`dt_0`。
332 if ((cur_time + dt_0) > (stop_time - eps))
333 dt_0 = stop_time - cur_time;
334 }
335
336 // 将调整后的`dt_0`重新分配到所有级别。
337 n_factor = 1; // 重置子循环因子。
338 for (int i = 0; i <= finest_level; i++) {
339 // 更新当前级别的子循环因子。
340 n_factor *= n_cycle[i];
341
342 // 为当前级别分配时间步长，按子循环因子缩放。
343 dt_level[i] = dt_0 / n_factor;
344 }
345}

350void AmrLevelCong::computeNewDt(int finest_level,
351 int /*sub_cycle*/,
352 Vector<int>& n_cycle,
353 const Vector<IntVect>& /*ref_ratio*/,
354 Vector<Real>& dt_min,
355 Vector<Real>& dt_level,
356 Real stop_time,
357 int post_regrid_flag)
358{
359 //
360 // 我们在粗网格时间循环的末尾。
361 // 计算下一次迭代的时间步长。
362 //
363 if (level > 0)
364 return;
365
366 for (int i = 0; i <= finest_level; i++) {
367 AmrLevelCong& adv_level = getLevel(i);
368 dt_min[i] = adv_level.estTimeStep(dt_level[i]);
369 }
370
371 if (post_regrid_flag == 1) {
372 //
373 // 按重网格前的dt限制dt
374 //
375 for (int i = 0; i <= finest_level; i++) {
376 dt_min[i] = amrex::min(dt_min[i], dt_level[i]);
377 }
378 } else {
379 //
380 // 按change_max * 旧dt限制dt
381 //
382 static Real change_max = 10;
383 for (int i = 0; i <= finest_level; i++) {
384 dt_min[i] = amrex::min(dt_min[i], change_max * dt_level[i]);
385 }
386 }
387
388 //
389 // 找到所有级别的最小值
390 //
391 Real dt_0 = 1.0e+100;
392 int n_factor = 1;
393 for (int i = 0; i <= finest_level; i++) {
394 n_factor *= n_cycle[i];
395 dt_0 = amrex::min(dt_0, n_factor * dt_min[i]);
396 }
397
398 //
399 // 按stop_time的值限制dt。
400 //
401 const Real eps = 0.001 * dt_0;
402 Real cur_time = state[State_Type].curTime();
403 if (stop_time >= 0.0) {
404 if ((cur_time + dt_0) > (stop_time - eps))
405 dt_0 = stop_time - cur_time;
406 }
407
408 n_factor = 1;
409 for (int i = 0; i <= finest_level; i++) {
410 n_factor *= n_cycle[i];
411 dt_level[i] = dt_0 / n_factor;
412 }
413}
