18EigenSystem::EigenSystem(VarArray const& primL, VarArray const& primR, GpuArray<Real, 3> norm, ProbParm const& pp)
19{
20 // 使用 Roe 平均计算中间状态
21 VarArray prim = average(primL, primR, norm, pp);
22
23 // 提取物理量
24 Real ar1 = prim[0]; // 第一相密度
25 Real ar2 = prim[1]; // 第二相密度
26 Real c = prim[2]; // 声速
27 Real r = prim[3]; // 总密度
28 Real alpha1 = prim[4]; // 第一相体积分数
29 Real nx = norm[0], ny = norm[1], nz = norm[2]; // 法向量分量
30
31 Real factor = nx + ny + nz;
32
33 // 初始化左特征矩阵
34#if BL_SPACEDIM == 1
35 Real ar2_over_c2r = ar2 / (c * c * r);
36 Real ar1_over_c2r = ar1 / (c * c * r);
37 Real half_over_cr = 1.0 / (2 * c * r);
38 leftEig = {
39 0, 0, 0, 0, 1.,
40 0, 1., 0, -ar2_over_c2r, 0,
41 1., 0, 0, -ar1_over_c2r, 0,
42 0, 0, -1. / 2.0, half_over_cr, 0,
43 0, 0, 1. / 2.0, half_over_cr, 0
44 };
45#elif BL_SPACEDIM == 2
46 Real ar2_over_c2r = ar2 / (c * c * r);
47 Real ar1_over_c2r = ar1 / (c * c * r);
48 Real over_cr = 1.0 / (c * r);
49 leftEig = {
50 0, 0, 0, 0, 0, 1,
51 0, 0, -ny, nx, 0, 0,
52 0, 1., 0, 0, -ar2_over_c2r, 0,
53 1.0, 0, 0, 0, -ar1_over_c2r, 0,
54 0, 0, -nx, -ny, over_cr, 0,
55 0, 0, nx, ny, over_cr, 0
56 };
57#elif BL_SPACEDIM == 3
58 // 下面三行是ai给注释时候自己加的，有待商榷
59 Real ar2_over_c2r = ar2 / (c * c * r);
60 Real ar1_over_c2r = ar1 / (c * c * r);
61 Real half_over_cr = 1.0 / (2 * c * r);
62 leftEig = {
63 0, 0, 0, 0, 0, 0, 1,
64 0, 0, nx * (nz - ny) / factor, (-ny * ny + ny * nz + 1) / factor, (-ny * nz + nz * nz - 1) / factor, 0, 0,
65 0, 0, (nx * nz + nz * nz + ny * ny) / factor, ny * (nz - nx) / factor, (-nx * nz + nz * nz - 1) / factor, 0, 0,
66 0, 1, 0, 0, 0, -ar2_over_c2r, 0,
67 1, 0, 0, 0, 0, -ar1_over_c2r, 0,
68 0, 0, -nx / 2.0, -ny / 2.0, -nz / 2.0, half_over_cr, 0,
69 0, 0, nx / 2.0, ny / 2.0, nz / 2.0, half_over_cr, 0
70 };
71#endif
72
73 // 初始化右特征矩阵
74#if BL_SPACEDIM == 1
75 rightEig = {
76 0, 0, 1., ar1 / c, ar1 / c,
77 0, 1., 0, ar2 / c, ar2 / c,
78 0, 0, 0, -1., 1.,
79 0, 0, 0, c * r, c * r,
80 1., 0, 0, 0, 0
81 };
82#elif BL_SPACEDIM == 2
83 rightEig = {
84 0, 0, 0, 1, ar1 / c / 2, ar1 / c / 2,
85 0, 0, 1, 0, ar2 / c / 2, ar2 / c / 2,
86 0, -ny, 0, 0, -nx / 2, nx / 2,
87 0, nx, 0, 0, -ny / 2, ny / 2,
88 0, 0, 0, 0, c * r / 2, c * r / 2,
89 1, 0, 0, 0, 0, 0
90 };
91#elif BL_SPACEDIM == 3
92 rightEig = {
93 0, 0, 0, 0, 1, ar1 / c, ar1 / c,
94 0, 0, 0, 1, 0, ar2 / c, ar2 / c,
95 0, -ny, ny + nz, 0, 0, -nx, nx,
96 0, nx + nz, -nx, 0, 0, -ny, ny,
97 0, -ny, -nx, 0, 0, -nz, nz,
98 0, 0, 0, 0, 0, c * r, c * r,
99 1, 0, 0, 0, 0, 0, 0
100 };
101#endif
102}

112AMREX_GPU_HOST_DEVICE
113VarArray EigenSystem::average(VarArray primL,
114 VarArray primR,
115 GpuArray<Real, 3> norm,
116 ProbParm const& pp) const
117{
118 // 提取左值
119 Real ar1L = primL[0], ar2L = primL[1], cL = eos_prim<SoundSpeed>(primL, pp), rL = ar1L + ar2L;
120
121 // 提取右值
122 Real ar1R = primR[0], ar2R = primR[1], cR = eos_prim<SoundSpeed>(primR, pp), rR = ar1R + ar2R;
123
124 // 计算算数平均
125 Real ar1_avg = 0.5 * (ar1L + ar1R);
126 Real ar2_avg = 0.5 * (ar2L + ar2R);
127 Real c_avg = 0.5 * (cL + cR);
128 Real r_avg = 0.5 * (rL + rR);
129 Real alpha1_avg = 0.5 * (primL[Alpha1] + primR[Alpha1]);
130
131 // 返回平均值
132 return { ar1_avg, ar2_avg, c_avg, r_avg, alpha1_avg };
133}
