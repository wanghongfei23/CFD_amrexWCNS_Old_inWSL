334// 计算侧面的theta值
335// 输入：维度idim，组件数ncomp，原始变量iPrim，守恒变量iCons，通量iFlux，一阶通量fluxFirstOrder，乘数mult，问题参数pp，保正函数pqf
336// 输出：theta值
337AMREX_GPU_HOST_DEVICE
338amrex::Real
339compute_theta_for_side(int idim, int ncomp,
340 const VarArray& iPrim, const VarArray& iCons,
341 const VarArray& iFlux,
342 const VarArray& fluxFirstOrder,
343 const amrex::Real& mult,
344 const ProbParm& pp,
345 const PositiveQuantityFunctions& pqf)
346{
347 VarArray consStar, consHLLC;
348 // 计算原始变量对应的通量
349 VarArray flux = prim_to_flux(iPrim, pp, get_direction(idim));
350
351 // 计算守恒变量的中间值
352 for (int ivar = 0; ivar < ncomp; ivar++) {
353 consStar[ivar] = iCons[ivar] - mult * (iFlux[ivar] - flux[ivar]);
354 consHLLC[ivar] = iCons[ivar] - mult * (fluxFirstOrder[ivar] - flux[ivar]);
355 }
356
357 // 计算theta值
358 Real theta = compute_theta(consStar, consHLLC, pp, pqf);
359 return theta;
360}

379// 计算theta值
380// 输入：守恒变量consStar，守恒变量consHLLC，问题参数pp，保正函数pqf
381// 输出：theta值
382AMREX_GPU_HOST_DEVICE
383amrex::Real
384compute_theta(VarArray const& consStar, VarArray const& consHLLC, ProbParm const& pp, PositiveQuantityFunctions const& pqf)
385{
386 // 计算theta_0的lambda函数
387 auto compute_theta_0 = [] AMREX_GPU_HOST_DEVICE(
388 amrex::Real positiveQuantity_W, // 原始变量中的正值
389 amrex::Real positiveQuantity_W_interpolated, // 插值后变量中的正值
390 amrex::Real EPSILON // 正值的阈值
391 ) -> amrex::Real {
392 return (EPSILON - positiveQuantity_W) / (positiveQuantity_W_interpolated - positiveQuantity_W);
393 };
394
395 // 计算保正变量值
396 auto pqStar = pqf(consStar, pp), pqHLLC = pqf(consHLLC, pp);
397 amrex::Real theta = 1.;
398
399 // 遍历所有保正变量
400 for (int ifunc = 0; ifunc < pqStar.size(); ifunc++) {
401 // 检查保正变量是否小于阈值
402 if (pqStar[ifunc] < pqf.eps[ifunc]) {
403 // 0.99的系数可以防止截断误差出问题
404 amrex::Real temptheta = compute_theta_0(pqHLLC[ifunc], pqStar[ifunc], pqf.eps[ifunc]);
405 theta = amrex::min(theta, temptheta * 0.99);
406 }
407 }
408
409 return theta;
410}
