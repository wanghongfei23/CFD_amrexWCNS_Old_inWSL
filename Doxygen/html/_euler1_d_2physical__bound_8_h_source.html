28 {
29 amrex::GpuArray<amrex::Real, 2> result; // 输出两个保正变量
30
31 // 第一个保正变量：密度

33
34 // 第二个保正变量：密度乘声速的平方

36 amrex::Real u = U[1] / rho; // x 方向速度
37 amrex::Real E = U[2]; // 总能量
38 amrex::Real p = (pp.gamma - 1.0) * (E - 0.5 * rho * u * u); // 压力公式
39 result[1] = pp.gamma * p; // 密度乘声速的平方
40
