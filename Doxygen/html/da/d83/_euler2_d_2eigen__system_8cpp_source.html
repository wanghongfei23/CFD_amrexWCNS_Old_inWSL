19EigenSystem::EigenSystem(VarArray const& primL, VarArray const& primR, GpuArray<Real, 3> norm, ProbParm const& pp)
20{
21 // 使用 Roe 平均计算中间状态
22 VarArray prim = average(primL, primR, norm, pp);
23 enum { X, Y };
24
25 // 提取物理量
26 Real gamma = pp.gamma; // 气体绝热指数
27 Real r = prim[0]; // 密度
28 Real u = prim[1]; // x方向速度
29 Real v = prim[2]; // y方向速度
30 Real p = prim[3]; // 压力
31 Real ek = (u * u + v * v) / 2; // 动能
32 Real h = p / r * gamma / (gamma - 1); // 比焓
33 Real c = sqrt(gamma * p / r); // 声速
34 Real ht = h + ek; // 总焓
35 Real Vn = u * norm[X] + v * norm[Y]; // 法向速度
36
37 // 计算左特征矩阵
38 leftEig = { (-norm[Y] * u + norm[X] * v),
39 norm[Y],
40 (-norm[X]),
41 0,
42 (h - ek),
43 u,
44 v,
45 -1.0,
46 (Vn / c + ek / h) / 2,
47 (-norm[X] / c - u / h) / 2,
48 (-norm[Y] / c - v / h) / 2,
49 1.0 / (2.0 * h),
50 (-Vn / c + ek / h) / 2,
51 (norm[X] / c - u / h) / 2,
52 (norm[Y] / c - v / h) / 2,
53 1.0 / (2.0 * h) };
54
55 // 计算右特征矩阵
56 rightEig = { 0,
57 1.0 / h,
58 1,
59 1,
60 norm[Y],
61 u / h,
62 u - norm[X] * c,
63 u + norm[X] * c,
64 -norm[X],
65 v / h,
66 v - norm[Y] * c,
67 v + norm[Y] * c,
68 norm[Y] * u - norm[X] * v,
69 ek / h,
70 h + ek - Vn * c,
71 h + ek + Vn * c };
72}

136AMREX_GPU_HOST_DEVICE
137VarArray EigenSystem::average(VarArray primL,
138 VarArray primR,
139 GpuArray<Real, 3> norm,
140 ProbParm const& pp) const
141{
142 Real gamma = pp.gamma; // 气体绝热指数
143 // 提取左右状态的物理量
144 Real rl = primL[0], ul = primL[1], vl = primL[2], pl = primL[3]; // 左侧：密度、x速度、y速度、压力
145 Real rr = primR[0], ur = primR[1], vr = primR[2], pr = primR[3]; // 右侧：密度、x速度、y速度、压力
146
147 enum { LL, RR };
148 GpuArray<Real, 2> H; // 总焓数组
149 H[LL] = (ul * ul + vl * vl) / 2 + pl / rl * gamma / (gamma - 1); // 左侧总焓
150 H[RR] = (ur * ur + vr * vr) / 2 + pr / rr * gamma / (gamma - 1); // 右侧总焓
151
152 // 计算加权系数
153 Real coef1 = sqrt(rl);
154 Real coef2 = sqrt(rr);
155 Real divisor = 1.0 / (coef1 + coef2);
156
157 // 计算平均物理量
158 Real r = sqrt(rl * rr); // 平均密度
159 Real u = (coef1 * ul + coef2 * ur) * divisor; // 平均x速度
160 Real v = (coef1 * vl + coef2 * vr) * divisor; // 平均y速度
161 Real Vn = norm[0] * u + norm[1] * v; // 平均法向速度
162 Real ht = (coef1 * H[LL] + coef2 * H[RR]) * divisor; // 平均总焓
163 Real ek = (u * u + v * v) / 2; // 平均动能
164 Real h = ht - ek; // 平均比焓
165 Real c = sqrt((gamma - 1) * h); // 平均声速
166 Real p = r * h * ((gamma - 1) / gamma); // 平均压力
167
168 return { r, u, v, p }; // 返回平均原始变量
169}
