30void runTestCase(
31 const std::string& testName,
32 const VarArray& cons,
33 const ProbParm& pp,
34 std::ostream& outFile,
35 int dimension = BL_SPACEDIM) // 问题的维度（1D, 2D, 3D）
36{

47 // 验证可逆性

51 // 检查重构守恒变量是否与原始守恒变量一致（容差 1e-5）
52 bool isConsEqual = true;
53 for (int i = 0; i < cons.size(); ++i) {
54 if (std::abs(cons[i] - cons_reconstructed[i]) > 1e-5) {
55 isConsEqual = false;

58 }

60 if (isConsEqual) {

66 // 状态方程计算
67 Real pressure = eos_cons<Pressure>(cons, pp);
68 Real total_energy = eos_prim<TotalEnergy>(prim, pp);
69 outFile << "Pressure from EOS: " << pressure << "\n";

71
72 // 根据维度测试不同的法向量 norm
73 std::vector<amrex::GpuArray<amrex::Real, 3>> norms;
74 if (dimension == 1) {
75 norms.push_back({ 1.0, 0.0, 0.0 }); // 1D: 只测试 norm = (1, 0, 0)
76 } else if (dimension == 2) {
77 norms.push_back({ 1.0, 0.0, 0.0 }); // 2D: 测试 norm = (1, 0, 0)

81 norms.push_back({ 0.0, 1.0, 0.0 }); // 和 norm = (0, 1, 0)
82 norms.push_back({ 0.0, 0.0, 1.0 }); // 和 norm = (0, 0, 1)
83 }
84
85 for (const auto& norm : norms) {

89 // 计算通量
90 VarArray flux = prim_to_flux(prim, pp, norm);
