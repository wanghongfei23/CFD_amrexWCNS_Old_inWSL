21AMREX_FORCE_INLINE bool
22computeTagValue(int i, int j, int k, amrex::Array4<amrex::Real const> const& S_new_array)
23{
24 bool res = false;
25
26 // Lambda函数，用于计算给定方向上的标记
27 auto computeFlag =
28 [&](int di, int dj, int dk)
29 -> AMREX_GPU_DEVICE bool {
30 // 提取当前点及其周围点的密度值
31 amrex::GpuArray<amrex::Real, 5> q = {
32 S_new_array(i + di * (-2), j + dj * (-2), k + dk * (-2), 0),
33 S_new_array(i + di * (-1), j + dj * (-1), k + dk * (-1), 0),
34 S_new_array(i, j, k, 0),
35 S_new_array(i + di * 1, j + dj * 1, k + dk * 1, 0),
36 S_new_array(i + di * 2, j + dj * 2, k + dk * 2, 0)
37 };
38 // 使用TENO5格式计算标记
39 return Teno5_CongZ_Flag(q);
40 };
41
42 // 计算各个维度上的标记
43 if constexpr (SpaceDim >= 1) {
44 res = res || computeFlag(1, 0, 0); // x方向
45 }
46 if constexpr (SpaceDim >= 2) {
47 res = res || computeFlag(0, 1, 0); // y方向
48 }
49 if constexpr (SpaceDim >= 3) {
50 res = res || computeFlag(0, 0, 1); // z方向
51 }
52
53 return res;
54}
