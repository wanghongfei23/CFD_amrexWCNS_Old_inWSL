25// 实施凸组合调整插值后的变量值
26// 输入：原始变量值W，高阶插值后的变量值W_interpolated，调整系数theta，起始组件索引scomp，组件的数量ncomp
27// 功能：根据theta值调整插值后的变量值
28AMREX_FORCE_INLINE AMREX_GPU_HOST_DEVICE void apply_convex_combination(
29 const VarArray& W, // 原始变量值
30 VarArray& W_interpolated, // 高阶插值后的变量值
31 amrex::Real theta, // 调整系数
32 int scomp, // 起始组件索引
33 int ncomp // 组件的数量
34)
35{
36 if (theta == 0.0 || contains_nan(W_interpolated)) {
37 for (int i = scomp; i < ncomp; ++i) {
38 W_interpolated[i] = W[i];
39 }
40 return;
41 }
42 for (int i = scomp; i < ncomp; ++i) {
43 W_interpolated[i] = (1.0 - theta) * W[i] + theta * W_interpolated[i];
44 }
45}

47// 检查变量数组是否包含无效值
48// 输入：变量数组varArray，问题参数pp，保正函数结构体pqf

50AMREX_INLINE AMREX_GPU_HOST_DEVICE bool containsInvalidValues(const VarArray& varArray,
51 const ProbParm& pp,
52 const PositiveQuantityFunctions& pqf)
53{

55
56 for (int ifunc = 0; ifunc < pq.size(); ++ifunc) {
57 amrex::Real value = pq[ifunc]; // 获取保正变量值
58
59 if (std::isnan(value) || !std::isfinite(value))
60 return true; // 检查是否为NaN或无穷大
61 if (value < pqf.eps[ifunc])
62 return true; // 检查是否小于阈值
63 }
64
65 return false;
