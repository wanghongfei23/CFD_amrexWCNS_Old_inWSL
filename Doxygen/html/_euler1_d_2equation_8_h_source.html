131// 守恒变量转换为原始变量
132// 输入：守恒变量cons，问题参数pp
133// 输出：原始变量
134AMREX_GPU_HOST_DEVICE
135AMREX_FORCE_INLINE
136VarArray cons_to_prim(VarArray cons, ProbParm const& pp)
137{
138 amrex::Real r, ru, rE;
139 r = cons[0]; // 密度
140 ru = cons[1]; // 动量
141 rE = cons[2]; // 总能量
142 amrex::Real u = ru / r; // 速度
143 amrex::Real E = rE / r; // 单位质量总能量
144 amrex::Real e = -u * u / 2 + E; // 单位质量内能
145
146 amrex::Real RT = (pp.gamma - 1) * e; // 气体常数乘温度
147 amrex::Real p = r * RT; // 压力
148 return { r, u, p };
149}

176// 原始变量转换为守恒变量
177// 输入：原始变量prim，问题参数pp
178// 输出：守恒变量
179AMREX_GPU_HOST_DEVICE
180AMREX_FORCE_INLINE
181VarArray prim_to_cons(VarArray prim, ProbParm const& pp)
182{
183 amrex::Real gamma = pp.gamma; // 气体绝热指数
184 amrex::Real rt = prim[0], ut = prim[1], pt = prim[2]; // 密度、速度、压力
185 amrex::Real ekt = (ut * ut) / 2; // 单位质量动能
186 amrex::Real rut = rt * ut; // 动量
187 amrex::Real ret = pt / (gamma - 1) + rt * ekt; // 总能量
188 return { rt, rut, ret };
189}

202// 原始变量转换为通量
203// 输入：原始变量prim，问题参数pp，法向量norm
204// 输出：通量
205 template <int idim = 0>
206AMREX_GPU_HOST_DEVICE
207 AMREX_FORCE_INLINE
208 VarArray
209 prim_to_flux(VarArray prim, ProbParm pp, amrex::GpuArray<amrex::Real, 3> norm = { 1, 0, 0 })
210{
211 amrex::Real r = prim[0], u = prim[1], p = prim[2]; // 密度、速度、压力
212 amrex::Real gamma = pp.gamma; // 气体绝热指数
213
214 // 计算通量
215 VarArray flux; // 通量：1. 密度通量，2. 动量通量，3. 能量通量
216 flux[0] = r * u;
217 flux[1] = r * u * u + p;
218 flux[2] = (gamma / (gamma - 1) * p + r * (u * u) / 2) * u;
219 // std::cout << flux[0] << ' ' << flux[1] << ' ' << flux[2] << '\n';
220
221 return flux;
222}

224// 计算最大特征速度
225// 输入：网格坐标i,j,k，状态数组state，最大速度数组umax，问题参数pp，网格间距dx

231 amrex::Array4<amrex::Real const> const& state,
232 amrex::Array4<amrex::Real> const& umax,
233 ProbParm const& pp, // 可选参数，支持不同气体模型
234 amrex::GpuArray<amrex::Real, BL_SPACEDIM> dx) noexcept
235{
236 amrex::Real gamma = pp.gamma; // 气体绝热指数
237 amrex::Real rho = state(i, j, k, 0); // 密度
238 // if (rho <= 1.e-10) {

240 // return;
241 // }
242 amrex::Real rhou = state(i, j, k, 1); // 动量

244
245 amrex::Real u = rhou / rho; // 速度
246 amrex::Real E = rhoE / rho; // 单位质量总能量
247 amrex::Real e = E - 0.5 * u * u; // 单位质量内能

250 amrex::Real p = rho * RT; // 压力
