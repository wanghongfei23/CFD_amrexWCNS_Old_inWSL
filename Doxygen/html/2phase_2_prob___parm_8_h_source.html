12struct ProbParm {
13 // 两相的比热比

22 // 计算平均比热比和参考压力
23 // 输入：第一相体积分数alpha1
24 // 输出：包含平均比热比和参考压力的数组
25 AMREX_FORCE_INLINE
26 AMREX_GPU_HOST_DEVICE
27 amrex::GpuArray<amrex::Real, 2> cal_gamma(amrex::Real alpha1) const
28 {
29 amrex::Real alpha2 = 1.0 - alpha1; // 第二相体积分数
30
31 // 混合区
32 amrex::Real gamma1 = gamma[0], gamma2 = gamma[1]; // 两相的比热比
33 amrex::Real pInf1 = pInf[0], pInf2 = pInf[1]; // 两相的参考压力
34
35 // 提前计算 gamma1 - 1 和 gamma2 - 1
36 amrex::Real gamma1_minus_1 = gamma1 - 1;
37 amrex::Real gamma2_minus_1 = gamma2 - 1;
38
39 // 计算平均比热比 \overline{\gamma}
40 amrex::Real inv_gamma_minus_1 = alpha1 / gamma1_minus_1 + alpha2 / gamma2_minus_1;
41 amrex::Real gamma_bar = 1.0 / inv_gamma_minus_1 + 1;
42
43 // 计算平均参考压力 \overline{p^\infty}
44 amrex::Real gamma_pInf_bar = alpha1 * gamma1 * pInf1 / gamma1_minus_1 + alpha2 * gamma2 * pInf2 / gamma2_minus_1;
45 amrex::Real pInf_bar = gamma_pInf_bar * (gamma_bar - 1) / gamma_bar;
46
47 return { gamma_bar, pInf_bar };
48 }
49};
