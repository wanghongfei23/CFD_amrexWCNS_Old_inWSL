198template <int RKOrder>
199constexpr auto get_RK_time_coefficients()
200{
201 static_assert(RKOrder >= 1 && RKOrder <= 4, "RKOrder must be 1, 2, 3, or 4.");
202 if constexpr (RKOrder == 1) {
203 return amrex::GpuArray<amrex::Real, 1> { 1.0 };
204 } else if constexpr (RKOrder == 2) {
205 return amrex::GpuArray<amrex::Real, 2> { 2.0, 2.0 };
206 } else if constexpr (RKOrder == 3) {
207 return amrex::GpuArray<amrex::Real, 3> { 6.0, 6.0, 3.0 / 2 };
208 } else if constexpr (RKOrder == 4) {
209 return amrex::GpuArray<amrex::Real, 4> { 0.0, 0.5, 0.5, 1.0 };
210 }

229{
230 // 获取旧数据，为通量保正限制器作准备
231 const MultiFab& old_data = get_old_data(State_Type);
232
233 // 创建原始变量数组
234 MultiFab prim(grids, dmap, NUM_STATE, NUM_GROW, MFInfo(), Factory());
235 for (MFIter mfi(Sborder, false); mfi.isValid(); ++mfi) {
236 Box ghost_box = amrex::grow(mfi.validbox(), NUM_GROW);
237 Array4<Real> const& prim_array = prim[mfi].array();
238 Array4<Real const> const& statein_array = Sborder[mfi].const_array();
239 ProbParm const& pp = *d_prob_parm;
240 // 将守恒变量转换为原始变量
241 cons_to_prim_for_array4(ghost_box, prim_array, statein_array, pp);
242 }
243
244 // 初始化时间导数为0
245 dSdt.setVal(0.0);
246 // 创建通量数组
247 MultiFab fluxes[BL_SPACEDIM];
248 // 获取网格间距
249 GpuArray<Real, BL_SPACEDIM> dx = geom.CellSizeArray();
250
251 // 如果需要回流，定义通量寄存器
252 if (do_reflux) {
253 for (int j = 0; j < BL_SPACEDIM; j++) {
254 BoxArray ba = prim.boxArray();
255 ba.surroundingNodes(j);
256 fluxes[j].define(ba, dmap, NUM_STATE, 0);
257 }
258 }
259#ifdef AMREX_USE_GPU
260 // 在GPU上，定义非守恒通量数组
261 MultiFab nonconservfluxes[BL_SPACEDIM];
262 for (int j = 0; j < BL_SPACEDIM; j++) {
263 BoxArray ba = prim.boxArray();
264 ba.surroundingNodes(j);
265 nonconservfluxes[j].define(ba, dmap, NUM_STATE * 2, 0);
266 }
267#endif
268 /*------------------------------------------------------------------------------*/
269#ifdef AMREX_USE_OMP
270#pragma omp parallel if (Gpu::notInLaunchRegion())
271#endif
272 {
273 // 守恒通量
274 amrex::GpuArray<FArrayBox, AMREX_SPACEDIM> fluxfab;
275 amrex::GpuArray<FArrayBox*, AMREX_SPACEDIM> flux;
276
277 // 非守恒通量
278 amrex::GpuArray<FArrayBox, AMREX_SPACEDIM> nonconserfluxfab;
279 amrex::GpuArray<FArrayBox*, AMREX_SPACEDIM> nonconserflux;

281 for (MFIter mfi(prim, false); mfi.isValid(); ++mfi) {
282 // 设置 tileboxes 和 nodal tileboxes

297#ifdef AMREX_USE_GPU
298 // 在 GPU 上不使用分块，直接将通量和面速度的 fab 指针指向未分块的 fab

300 nonconserflux[idim] = &(nonconservfluxes[idim][mfi]);
301#else
302 // 在 CPU 上，重新调整临时 fab 的大小以存储通量和面速度
303 const Box& bxflux = amrex::surroundingNodes(bx, idim);
304 fluxfab[idim].resize(bxflux, NUM_STATE);
305
306 // 将通量和面速度的 fab 指针指向临时 fab
307 flux[idim] = &(fluxfab[idim]);
308
309 // 处理非守恒通量，只在mfi的局部使用
310 nonconserfluxfab[idim].resize(bxflux, NUM_STATE * 2);
311 nonconserflux[idim] = &(nonconserfluxfab[idim]);
312#endif
313 }

318 Real RKTimeCoef = get_RK_time_coefficients<RKOrder>()[stage - 1];
319 // 使用 AMREX_D_TERM 宏遍历所有空间维度，计算各维度的通量
320 AMREX_D_TERM(
321 flux_solve_in_dim<0>({ bx, flux, nonconserflux, cellFlux, cellNcFlux, primin, consin, rhs, nbx, dt, RKTimeCoef });,
322 flux_solve_in_dim<1>({ bx, flux, nonconserflux, cellFlux, cellNcFlux, primin, consin, rhs, nbx, dt, RKTimeCoef });
323 ,
324 flux_solve_in_dim<2>({ bx, flux, nonconserflux, cellFlux, cellNcFlux, primin, consin, rhs, nbx, dt, RKTimeCoef });)
325
326 // 通量保正限制器（当前被注释掉）
327 // 获取当前 Runge - Kutta 阶段的时间系数
328 // Real RKTimeCoef = get_RK_time_coefficients<RKOrder>()[stage - 1];
329
330 // ProbParm const& pp = *d_prob_parm;
331
332 // // // 定义正限制函数对象
333 // const PositiveQuantityFunctions pqf;
334 // const FArrayBox& old_data_fab = old_data[mfi];
335
336 // apply_positivive_preserving_flux_limiter(
337 // bx, // 计算区域
338 // Sborder[mfi], // 守恒量数组
339 // primin, // 原始量数组

385 fine->CrseInit(fluxes[i], i, 0, 0, NUM_STATE, -fluxScalingFactors[i], FluxRegister::ADD);
386 }
387 }

395template <int idim>
396void AmrLevelCong::flux_solve_in_dim(const FluxSolveParams& params)
397{
398 // 解包参数
399 const auto& bx = params.bx;
400 auto& flux = params.flux;
401 auto& ncflux = params.ncflux;
402 auto& cellFlux = params.cellFlux;
403 auto& cellNcFlux = params.cellNcFlux;
404 const auto& primin = params.primin;
405 const auto& consin = params.consin;
406 auto& rhs = params.rhs;
407 const auto& nbx = params.nbx;
408 const auto& dt = params.dt;
409 const auto& RKTimeCoef = params.RKTimeCoef;
410
411 // 断言通量计算区域包含计算盒
412 AMREX_ASSERT(cellFlux.box().contains(bx));
413 AMREX_ASSERT(cellNcFlux.box().contains(bx));
414
415 // 维度偏移量（用于跨维度访问）
416 Dim3 offset = amrex::IntVect::TheDimensionVector(idim).dim3();
417
418 // 获取问题参数和几何信息
419 ProbParm const& pp = *d_prob_parm;
420 amrex::GpuArray<amrex::Real, 3> norm = get_direction<idim>();
421 GpuArray<Real, BL_SPACEDIM> dx = geom.CellSizeArray();
422
423 // 扩展通量计算区域（包含虚拟网格）
424 Box fluxBx = amrex::grow(bx, num_flux_ghost_cells);
425
426 // 初始化节点通量存储
427 cellFlux.setVal<RunOn::Device>(0.0);
428
429 // 初始化非守恒通量
430 cellNcFlux.setVal<RunOn::Device>(0.0);
431 ncflux[idim]->setVal<RunOn::Device>(0.0);
432
433 // 获取数组视图
434 Array4<Real const> const& primin_array = primin.const_array();
435 Array4<Real const> const& consin_array = consin.const_array();
436 Array4<Real> const& fluxRef = flux[idim]->array();
437 Array4<Real> const& stateout_array = rhs.array();
438 Array4<Real> const& nodeFlux_array = cellFlux.array();
439
440 // 非守恒通量数组视图
441 Array4<Real> const& nodencFlux_array = cellNcFlux.array();
442 Array4<Real> const& ncfluxRef = ncflux[idim]->array();
443
444 // ----------------------------
445 // 阶段1：计算节点通量
446 // ----------------------------
447 ParallelFor(fluxBx, [=] AMREX_GPU_DEVICE(int xIdx, int yIdx, int zIdx) {
448 // 提取原始状态变量
449 VarArray primitiveState;
450 for (int stateIdx = 0; stateIdx < NUM_STATE; stateIdx++) {
451 primitiveState[stateIdx] = primin_array(xIdx, yIdx, zIdx, stateIdx);
452 }
453
454 // 计算守恒通量
455 VarArray fluxState = prim_to_flux(primitiveState, pp, norm);
456 for (int stateIdx = 0; stateIdx < NUM_STATE; stateIdx++) {
457 nodeFlux_array(xIdx, yIdx, zIdx, stateIdx) = fluxState[stateIdx];
458 }
459
460 // 计算非守恒项通量
461 AdvectionTerm advTerm;
462 auto flux_var = advTerm.computeNodeFlux(primitiveState, pp, norm);
463 for (int idx = 0; idx < AdvectionTerm::num_terms; idx++) {
464 nodencFlux_array(xIdx, yIdx, zIdx, idx) += flux_var[idx];
465 }
466 });
467
468 // ----------------------------
469 // 阶段2：Riemann求解器计算通量
470 // ----------------------------
471 ParallelFor(nbx[idim], [=] AMREX_GPU_DEVICE(int i, int j, int k) {
472 // 重构左右状态
473 amrex::GpuArray<VarArray, 2> stateLR = reconstruction<idim>(i, j, k, primin_array, NUM_STATE, pp, norm, dx);
474
475 // 使用 Riemann 求解器计算通量
476 VarArray iFlux = Riemann_solver<idim>(i, j, k, primin_array, stateLR[0], stateLR[1], norm, pp, dx);
477
478 // 对计算出的通量进行修正，使用四阶精度的中心差分格式
479 VarArray consL, consR, fluxCellL, fluxCellR;
480 for (int ivar = 0; ivar < NUM_STATE; ivar++) {
481 consL[ivar] = consin_array(i - offset.x, j - offset.y, k - offset.z, ivar); // 左侧单元的守恒变量
482 consR[ivar] = consin_array(i, j, k, ivar); // 右侧单元的守恒变量
483 fluxCellL[ivar] = nodeFlux_array(i - offset.x, j - offset.y, k - offset.z, ivar); // 左侧单元的通量
484 fluxCellR[ivar] = nodeFlux_array(i, j, k, ivar); // 右侧单元的通量
485 }
486
487 // 通量修正
488 for (int ivar = 0; ivar < NUM_STATE; ivar++) {
489 iFlux[ivar] = 4.0 / 3.0 * iFlux[ivar]
490 - 1.0 / 6.0 * (nodeFlux_array(i - offset.x, j - offset.y, k - offset.z, ivar) + nodeFlux_array(i, j, k, ivar));
491 }

495 // 计算 alpha du/dx 中半节点处的 u_1/2
496 auto flux_var = advTerm.computeFlux(stateLR[0], stateLR[1], pp, norm, dx);
497 // 获取输运项对应的方程索引
498 auto indices = advTerm.getTargetTermIndices();

501 VarArray primL, primR;
502 for (int ivar = 0; ivar < NUM_STATE; ivar++) {
503 primL[ivar] = primin_array(i - offset.x, j - offset.y, k - offset.z, ivar); // 左侧单元的原始变量
504 primR[ivar] = primin_array(i, j, k, ivar); // 右侧单元的原始变量
505 }
506 // 使用 AdvectionTerm 计算单元中心处的系数 alpha
507 auto coefL = advTerm.computeCoefficient(primL, pp, norm);
508 auto coefR = advTerm.computeCoefficient(primR, pp, norm);
509
510 // 初始化增量通量数组
511 VarArray incFluxL, incFluxR;
512 incFluxL.fill(0.0);
513 incFluxR.fill(0.0);
514
515 // 根据非守恒项计算增量通量
516 for (int idx = 0; idx < AdvectionTerm::num_terms; idx++) {
517 incFluxL[indices[idx]] += coefL[idx]
518 * (4.0 / 3.0 * flux_var[idx]
519 - 1.0 / 6.0 * (nodencFlux_array(i - offset.x, j - offset.y, k - offset.z, idx) + nodencFlux_array(i, j, k, idx)));
520 incFluxR[indices[idx]] += coefR[idx]
521 * (4.0 / 3.0 * flux_var[idx]
522 - 1.0 / 6.0 * (nodencFlux_array(i - offset.x, j - offset.y, k - offset.z, idx) + nodencFlux_array(i, j, k, idx)));
523 }
524
525 // 计算时间步长相关的乘数因子
526 Real multL = 2. * dt * RKTimeCoef / dx[idim] / computeAlpha(dx, primL, pp)[idim];
527 Real multR = -2. * dt * RKTimeCoef / dx[idim] / computeAlpha(dx, primR, pp)[idim];
528
529 // 计算中间状态的守恒变量
530 VarArray consStarL, consStarR;
531 PositiveQuantityFunctions pqf;
532 for (int ivar = 0; ivar < NUM_STATE; ivar++) {
533 consStarL[ivar] = consL[ivar] - multL * (iFlux[ivar] + incFluxL[ivar] - fluxCellL[ivar]);
534 consStarR[ivar] = consR[ivar] - multR * (iFlux[ivar] + incFluxR[ivar] - fluxCellR[ivar]);
535 }
536
537 // 如果中间状态包含无效值，则应用凸组合限制器
538 bool leftFlag = containsInvalidValues(consStarL, pp, pqf);
539 bool rightFlag = containsInvalidValues(consStarR, pp, pqf);
540 if (leftFlag || rightFlag) {
541 VarArray foFlux = Riemann_solver2(i, j, k, primin_array, primL, primR, norm, pp), foncFluxL, foncFluxR;
542 foncFluxL.fill(0.0);
543 foncFluxR.fill(0.0);
544 auto foflux_var = advTerm.computeFlux(primL, primR, pp, norm, dx);
545 for (int idx = 0; idx < AdvectionTerm::num_terms; idx++) {
546 foncFluxL[indices[idx]] += coefL[idx] * foflux_var[idx];
547 foncFluxR[indices[idx]] += coefR[idx] * foflux_var[idx];
548 }
549 // 计算限制后的守恒变量
550 VarArray consFoL, consFoR;
551 for (int ivar = 0; ivar < NUM_STATE; ivar++) {
552 consFoL[ivar] = consL[ivar] - multL * (foFlux[ivar] + foncFluxL[ivar] - fluxCellL[ivar]);
553 consFoR[ivar] = consR[ivar] - multR * (foFlux[ivar] + foncFluxR[ivar] - fluxCellR[ivar]);
554 }
555 // 使用 Lambda 表达式计算 thetaL 和 thetaR
556 auto computeThetaWithCondition = [](bool flag, Real (*computeThetaFunc)(const VarArray&, const VarArray&, const ProbParm&, const PositiveQuantityFunctions&),
557 const VarArray& consStar, const VarArray& consFo, const ProbParm& pp, const PositiveQuantityFunctions& pqf) -> Real {
558 return flag ? computeThetaFunc(consStar, consFo, pp, pqf) : 1.0;
559 };
560 Real thetaL = computeThetaWithCondition(leftFlag, compute_theta, consStarL, consFoL, pp, pqf);
561 Real thetaR = computeThetaWithCondition(rightFlag, compute_theta, consStarR, consFoR, pp, pqf);
562 Real theta = min(thetaL, thetaR);
563 apply_convex_combination(foFlux, iFlux, theta, 0, NVar);
564 apply_convex_combination(foncFluxL, incFluxL, theta, 0, NVar);
565 apply_convex_combination(foncFluxR, incFluxR, theta, 0, NVar);
566 }
567
568 // 更新通量和非守恒通量数组
569 for (int ivar = 0; ivar < NVar; ivar++) {
570 fluxRef(i, j, k, ivar) = iFlux[ivar]; // 更新守恒通量
571 ncfluxRef(i, j, k, ivar) = incFluxL[ivar]; // 更新左侧非守恒通量
572 ncfluxRef(i, j, k, ivar + NVar) = incFluxR[ivar]; // 更新右侧非守恒通量
573 }
574 });
575
576 // ----------------------------
577 // 阶段3：更新右手边（RHS）
578 // ----------------------------
579 ParallelFor(bx, [=] AMREX_GPU_DEVICE(int i, int j, int k) {
580 for (int ivar = 0; ivar < NUM_STATE; ivar++) {
581 // 守恒项通量差分
582 Real flux_diff = (fluxRef(i + offset.x, j + offset.y, k + offset.z, ivar) - fluxRef(i, j, k, ivar)) / dx[idim];
583 stateout_array(i, j, k, ivar) -= flux_diff;
584
585 // 非守恒项差分
586 // 注意要用右面的左值减去左面的右值
587 Real ncflux_diff = (ncfluxRef(i + offset.x, j + offset.y, k + offset.z, ivar) - ncfluxRef(i, j, k, ivar + NUM_STATE)) / dx[idim];
588 stateout_array(i, j, k, ivar) -= ncflux_diff;
589 }
