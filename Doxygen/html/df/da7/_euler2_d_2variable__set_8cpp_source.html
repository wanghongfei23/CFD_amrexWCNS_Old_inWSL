19void AmrLevelCong::variableSetUp()
20{
21 BL_ASSERT(desc_lst.size() == 0); // 确保描述符列表为空
22
23 // 初始化包含问题特定变量的结构体
24 h_prob_parm = new ProbParm {}; // 主机端问题参数
25 d_prob_parm = (ProbParm*)The_Arena()->alloc(sizeof(ProbParm)); // 设备端问题参数
26 // 将主机端参数复制到设备端
27 amrex::Gpu::copy(amrex::Gpu::hostToDevice, h_prob_parm, h_prob_parm + 1, d_prob_parm);
28
29 // 读取参数，设置物理边界条件
30 read_params();
31
32 // 添加状态描述符
33 desc_lst.addDescriptor(State_Type, IndexType::TheCellType(),
34 StateDescriptor::Point, NUM_GROW, NUM_STATE,
35 &cell_cons_interp); // 使用守恒变量插值方法
36
37 // 使用ParmParse读取每个变量的边界条件
38 ParmParse pp("amr");
39
40 // 读取边界条件的辅助函数
41 auto readBC = [&pp](const std::string& prefix) -> BCRec {
42 int lo_bc[BL_SPACEDIM]; // 下界边界条件
43 int hi_bc[BL_SPACEDIM]; // 上界边界条件
44 for (int i = 0; i < BL_SPACEDIM; ++i) {
45 // 动态生成边界条件名称
46 std::string lo_bc_name = prefix + "_lo_bc_" + std::to_string(i);
47 std::string hi_bc_name = prefix + "_hi_bc_" + std::to_string(i);
48
49 // 如果未指定，默认为外推边界条件
50 pp.query(lo_bc_name.c_str(), lo_bc[i]);
51 pp.query(hi_bc_name.c_str(), hi_bc[i]);
52
53 if (!pp.contains(lo_bc_name.c_str())) {
54 lo_bc[i] = BCType::foextrap;
55 }
56 if (!pp.contains(hi_bc_name.c_str())) {
57 hi_bc[i] = BCType::foextrap;
58 }
59 }
60 return BCRec(lo_bc, hi_bc);
61 };
62
63 // 读取每个变量的边界条件
64 BCRec density_bc = readBC("density"); // 密度边界条件
65 BCRec xvelocity_bc = readBC("xvelocity"); // x方向速度边界条件
66 BCRec yvelocity_bc = readBC("yvelocity"); // y方向速度边界条件
67 BCRec energy_bc = readBC("energy"); // 能量边界条件
68
69 // 设置边界函数
70 StateDescriptor::BndryFunc bndryfunc(nullfill);
71 bndryfunc.setRunOnGPU(true); // 确保边界函数在GPU上运行
72
73 // 设置状态变量组件
74 desc_lst.setComponent(State_Type, 0, "Density", density_bc, bndryfunc); // 密度
75 desc_lst.setComponent(State_Type, 1, "XVelocity", xvelocity_bc, bndryfunc); // x方向速度
76 desc_lst.setComponent(State_Type, 2, "YVelocity", yvelocity_bc, bndryfunc); // y方向速度
77 desc_lst.setComponent(State_Type, 3, "TotalEnergy", energy_bc, bndryfunc); // 总能量
78}
