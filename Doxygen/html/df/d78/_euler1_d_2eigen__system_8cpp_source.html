19EigenSystem::EigenSystem(VarArray const& primL, VarArray const& primR, GpuArray<Real, 3> norm, ProbParm const& pp)
20{
21 // 计算左右状态的平均
22 VarArray prim = average(primL, primR, norm, pp);
23
24 // 提取平均状态的物理量
25 Real gamma = pp.gamma; // 气体绝热指数
26 Real r = prim[0]; // 密度
27 Real u = prim[1]; // 速度
28 Real p = prim[2]; // 压力
29 Real ek = (u * u) / 2; // 动能
30 Real h = p / r * gamma / (gamma - 1); // 比焓
31 Real c = sqrt(gamma * p / r); // 声速
32 Real ht = h + ek; // 总焓
33
34 // 计算左右特征矩阵
35 Real gamma_1 = gamma - 1;
36 // 左特征矩阵
37 leftEig = { ht + c * (u - c) / gamma_1, -u - c / gamma_1, 1.0,
38 -2.0 * ht + 4.0 * c * c / gamma_1, 2.0 * u, -2.0,
39 ht - c * (u + c) / gamma_1, -u + c / gamma_1, 1.0 };
40 Real factorEig = 0.5 * gamma_1 / (c * c); // 归一化因子
41
42 // 归一化左特征矩阵
43 for (int ii = 0; ii < 9; ii++)
44 leftEig[ii] *= factorEig;
45
46 // 右特征矩阵
47 rightEig = { 1.0, 1.0, 1.0, u - c, u,
48 u + c, ht - u * c, 0.5 * u * u, ht + u * c };
49}

113AMREX_GPU_HOST_DEVICE
114VarArray EigenSystem::average(VarArray primL,
115 VarArray primR,
116 GpuArray<Real, 3> norm,
117 ProbParm const& pp) const
118{
119 Real gamma = pp.gamma; // 气体绝热指数
120 // 提取左右状态的物理量
121 Real rl = primL[0], ul = primL[1], pl = primL[2]; // 左侧：密度、速度、压力
122 Real rr = primR[0], ur = primR[1], pr = primR[2]; // 右侧：密度、速度、压力
123
124 enum { LL, RR };
125 GpuArray<Real, 2> H; // 总焓数组
126 H[LL] = (ul * ul) / 2 + pl / rl * gamma / (gamma - 1); // 左侧总焓
127 H[RR] = (ur * ur) / 2 + pr / rr * gamma / (gamma - 1); // 右侧总焓
128
129 // 计算加权系数
130 Real coef1 = sqrt(rl);
131 Real coef2 = sqrt(rr);
132 Real divisor = 1.0 / (sqrt(rl) + sqrt(rr));
133
134 // 计算平均物理量
135 Real r = sqrt(rl * rr); // 平均密度
136 Real u = (coef1 * ul + coef2 * ur) * divisor; // 平均速度
137 Real ht = (coef1 * H[LL] + coef2 * H[RR]) * divisor; // 平均总焓
138 Real ek = (u * u) / 2; // 平均动能
139 Real h = ht - ek; // 平均比焓
140 Real c = sqrt((gamma - 1) * h); // 平均声速
141 Real p = r * h * ((gamma - 1) / gamma); // 平均压力
142
143 return { r, u, p }; // 返回平均原始变量
144}
