25// 检查并打印NaN值的函数
26// 输入：Amr对象amr
27// 输出：是否检测到NaN值的布尔值
28bool CheckForNaNAndPrint(Amr& amr)
29{
30 bool nanDetected = false;
31
32 // 遍历所有层级
33 for (int level = 0; level <= amr.finestLevel(); ++level) {
34 const MultiFab& state = amr.getLevel(level).get_new_data(0); // 获取当前层级的状态数据
35 const Geometry& geom = amr.Geom(level); // 获取几何对象
36
37 // 使用 MFIter 遍历网格
38 for (MFIter mfi(state); mfi.isValid(); ++mfi) {
39 const Box& box = mfi.validbox();
40 const FArrayBox& fab = state[mfi]; // 获取当前网格的数据
41
42 // 遍历所有变量分量
43 for (int comp = 0; comp < fab.nComp(); ++comp) {
44 // 使用 ParallelFor 并行遍历 Box 中的单元
45 amrex::LoopOnCpu(box, [&](int i, int j, int k) {
46 IntVect iv(AMREX_D_DECL(i, j, k));
47 if (std::isnan(fab(iv, comp))) { // 检查是否为 NaN
48 // 计算物理位置
49 amrex::Real physLoc[AMREX_SPACEDIM];
50 geom.CellCenter(iv, physLoc);
51
52 // 直接打印信息
53 std::cout << "NaN detected at the following location:\n";
54 std::cout << " Simulation step: " << amr.levelSteps(0)
55 << ", Simulation time: " << amr.cumTime() << "\n";
56 std::cout << " Level: " << level << ", Index: " << iv
57 << ", Component: " << comp << "\n";
58 std::cout << " Physical Location: (";
59 for (int d = 0; d < AMREX_SPACEDIM; ++d) {
60 std::cout << physLoc[d];
61 if (d < AMREX_SPACEDIM - 1) {
62 std::cout << ", ";
63 }
64 }
65 std::cout << ")\n";
66 nanDetected = true;
67 }
68 });
69 }
70 }
71 }
72
73 // 强制刷新输出缓冲区
74 std::cout << std::flush;
75
76 return nanDetected;
77}
