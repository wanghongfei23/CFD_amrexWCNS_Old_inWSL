14// 状态方程类型枚举
15enum EOSType {
16 Pressure = 0, // 压力
17 SoundSpeed, // 声速
18 SoundSpeedSqrDensity, // 声速平方乘以密度
19 TotalEnergy // 总能量
20};
21// 从守恒变量计算状态方程相关量
22// 输入：守恒变量cons，问题参数pp
23// 输出：根据eType返回相应的状态量（压力、声速等）
24template <EOSType eType>
25AMREX_GPU_HOST_DEVICE
26 AMREX_FORCE_INLINE
27 amrex::Real
28 eos_cons(VarArray cons, ProbParm const& pp)
29{
30 using namespace amrex;
31 Real rho1 = cons[AlphaRho1], rho2 = cons[AlphaRho2]; // 两相密度
32 Real AMREX_D_DECL(rhou = cons[XMom], rhov = cons[YMom], rhow = cons[YMom]); // 动量分量
33 Real rhoE = cons[Energy], alpha1 = cons[Alpha1]; // 总能量和相分数
34
35 Real halfRhoU_sqr = (AMREX_D_TERM(rhou * rhou, +rhov * rhov, +rhow * rhow)) / 2; // 动能
36
37 // Lambda 函数：根据 EOSType 返回相应的值
38 auto computeState = [&](Real gamma, Real pInf, Real rho) -> Real {
39 Real internalEnergy = rhoE - halfRhoU_sqr / rho; // 内能
40 Real pressure = (gamma - 1) * internalEnergy - gamma * pInf; // 压力
41 Real soundSpeedSqrDensity = gamma * (gamma - 1) * (internalEnergy - pInf); // 声速平方乘以密度
42 Real soundSpeed = sqrt(soundSpeedSqrDensity / rho); // 声速
43
44 if constexpr (eType == Pressure) {
45 return pressure;
46 } else if constexpr (eType == SoundSpeed) {
47 return soundSpeed;
48 } else if constexpr (eType == SoundSpeedSqrDensity) {
49 return soundSpeedSqrDensity;
50 }
51 };
52
53 // 使用 cal_gamma 全部用混合区的逻辑进行计算
54 amrex::GpuArray<amrex::Real, 2> gamma_pInf = pp.cal_gamma(alpha1); // 计算混合区的gamma和pInf
55 Real gamma_bar = gamma_pInf[0]; // 混合区的绝热指数
56 Real pInf_bar = gamma_pInf[1]; // 混合区的参考压力
57 // 计算混合区状态
58 // Real rho = (alpha1 < pp.alphaEps) ? rho2 : ((1 - alpha1) < pp.alphaEps) ? rho1
59 // : (rho1 + rho2);
60 Real rho = rho1 + rho2; // 总密度
61 return computeState(gamma_bar, pInf_bar, rho);
62}

64// 从原始变量计算状态方程相关量
65// 输入：原始变量prim，问题参数pp
66// 输出：根据eType返回相应的状态量（总能量、声速等）
67template <EOSType eType>
68AMREX_GPU_HOST_DEVICE
69 AMREX_FORCE_INLINE
70 amrex::Real
71 eos_prim(const VarArray& prim, ProbParm const& pp)
72{
73 using namespace amrex;
74
75 // 提取原始变量
76 Real rho1 = prim[AlphaRho1], rho2 = prim[AlphaRho2]; // 两相密度
77 Real AMREX_D_DECL(u = prim[XMom], v = prim[YMom], w = prim[ZMom]); // 速度分量
78 Real pressure = prim[Energy], alpha1 = prim[Alpha1]; // 压力和相分数
79
80 // 计算混合密度
81 // Real rho = (alpha1 < pp.alphaEps) ? rho2 : ((1 - alpha1) < pp.alphaEps) ? rho1
82 // : (rho1 + rho2);
83 Real rho = rho1 + rho2; // 总密度
84 // 计算动能
85 Real kineticEnergy = rho * (AMREX_D_TERM(u * u, +v * v, +w * w)) / 2; // 动能
86
87 // Lambda 函数：根据 EOSType 计算输出
88 auto computeOutput = [&](Real gamma, Real pInf, Real rho, Real pressure) -> Real {
89 if constexpr (eType == EOSType::TotalEnergy) {
90 // 计算总能量
91 Real internalEnergy = (pressure + gamma * pInf) / (gamma - 1); // 内能
92 return internalEnergy + kineticEnergy; // 总能量 = 内能 + 动能
93 } else if constexpr (eType == EOSType::SoundSpeed) {
94 // 计算声速
95 Real soundSpeedSqr = gamma * (pressure + pInf) / rho; // 声速平方
96 return sqrt(soundSpeedSqr); // 声速
97 } else if constexpr (eType == EOSType::SoundSpeedSqrDensity) {
98 // 计算声速平方乘以密度
99 return gamma * (pressure + pInf);
100 }
101 };
102
103 // 混合区
104 amrex::GpuArray<amrex::Real, 2> gamma_pInf = pp.cal_gamma(alpha1); // 计算混合区的gamma和pInf
105 Real gamma_bar = gamma_pInf[0]; // 混合区的绝热指数
106 Real pInf_bar = gamma_pInf[1]; // 混合区的参考压力
107
108 // 计算混合区输出
109 return computeOutput(gamma_bar, pInf_bar, rho, pressure);
110}

112// 守恒变量转换为原始变量
113// 输入：守恒变量cons，问题参数pp
114// 输出：原始变量
115AMREX_GPU_HOST_DEVICE
116AMREX_FORCE_INLINE
117VarArray cons_to_prim(VarArray cons, ProbParm const& pp)
118{
119 using namespace amrex;
120
121 // 守恒变量
122 Real rho1 = cons[AlphaRho1], rho2 = cons[AlphaRho2]; // 两相密度
123 Real AMREX_D_DECL(rhou = cons[XMom], rhov = cons[YMom], rhow = cons[ZMom]); // 动量分量
124 Real rhoE = cons[Energy], alpha1 = cons[Alpha1]; // 总能量和相分数
125
126 // 原始变量
127 // Real rho = (alpha1 < pp.alphaEps) ? rho2 : ((1 - alpha1) < pp.alphaEps) ? rho1
128 // : (rho1 + rho2);
129 Real rho = rho1 + rho2; // 总密度
130 Real AMREX_D_DECL(u = rhou / rho, v = rhov / rho, w = rhow / rho); // 速度分量
131
132 // 调用 state_of_equation 计算压力
133 Real pressure = eos_cons<Pressure>(cons, pp); // 计算压力
134
135 // 返回原始变量
136 return { rho1, rho2, AMREX_D_DECL(u, v, w), pressure, alpha1 };
137}

140// 输入：原始变量prim，问题参数pp

143AMREX_FORCE_INLINE
144VarArray prim_to_cons(VarArray prim, ProbParm const& pp)
145{
146 using namespace amrex;
147
148 // 从原始变量中提取值
149 Real rho1 = prim[AlphaRho1], rho2 = prim[AlphaRho2]; // 两相密度
150 Real AMREX_D_DECL(u = prim[XMom], v = prim[YMom], w = prim[ZMom]); // 速度分量
151 Real pressure = prim[Pressure]; // 压力
152 Real alpha1 = prim[Alpha1]; // 相分数
153

194// 原始变量转换为通量
195// 输入：原始变量prim，问题参数pp，法向量norm
196// 输出：通量
197AMREX_GPU_HOST_DEVICE
198AMREX_FORCE_INLINE
199VarArray
200prim_to_flux(const VarArray& prim, ProbParm const& pp, amrex::GpuArray<amrex::Real, 3> norm)
201{
202 using namespace amrex;
203
204 // 提取原始变量
205 Real rho1 = prim[AlphaRho1], rho2 = prim[AlphaRho2]; // 两相密度
206 Real AMREX_D_DECL(u = prim[XMom], v = prim[YMom], w = prim[ZMom]); // 速度分量
207 Real pressure = prim[Energy], alpha1 = prim[Alpha1]; // 压力和相分数
208
209 // 计算混合密度
210 // Real rho = (alpha1 < pp.alphaEps) ? rho2 : ((1 - alpha1) < pp.alphaEps) ? rho1
211 // : (rho1 + rho2);
212 Real rho = rho1 + rho2; // 总密度
213
214 // 计算法向速度 Vn
215 Real Vn = AMREX_D_TERM(u * norm[0], +v * norm[1], +w * norm[2]); // 法向速度
216
217 // 计算总能量 rhoE
218 Real rhoE = eos_prim<TotalEnergy>(prim, pp); // 总能量
219
220 // 计算通量
221 VarArray flux = {
222 rho1 * Vn, // 相1密度通量
223 rho2 * Vn, // 相2密度通量
224 AMREX_D_DECL(rho * u * Vn + pressure * norm[0], // 动量通量
225 rho * v * Vn + pressure * norm[1],
226 rho * w * Vn + pressure * norm[2]),
227 (rhoE + pressure) * Vn, // 能量通量
228 0. // 相分数通量
229 };
230
231 return flux;
232}

234// 计算法向量
235// 输入：网格坐标i,j,k，原始变量数组primin_array，网格间距dx
236// 输出：归一化的法向量
237AMREX_GPU_HOST_DEVICE
238AMREX_FORCE_INLINE
239amrex::GpuArray<amrex::Real, 3>
240compute_normal_vector1(int i, int j, int k,
241 amrex::Array4<amrex::Real const> const& primin_array,
242 amrex::GpuArray<amrex::Real, AMREX_SPACEDIM> dx)
243{
244 if constexpr (BL_SPACEDIM == 1) {
245 return { 1.0, 0.0, 0.0 }; // 一维情况下直接返回x方向法向量
246 }
247 // 使用数组存储梯度分量，初始化为 0.0
248 amrex::Real dpsi[3] = { 0.0 };
249
250 // 计算梯度分量，使用 AMREX_D_TERM 宏适配不同维度
251 AMREX_D_TERM(
252 dpsi[0] = (primin_array(i + 1, j, k, Alpha1) - primin_array(i - 1, j, k, Alpha1)) / (2.0 * dx[0]);, // x方向梯度
253 dpsi[1] = (primin_array(i, j + 1, k, Alpha1) - primin_array(i, j - 1, k, Alpha1)) / (2.0 * dx[1]); // y方向梯度
254 ,
255 dpsi[2] = (primin_array(i, j, k + 1, Alpha1) - primin_array(i, j, k - 1, Alpha1)) / (2.0 * dx[2]);); // z方向梯度
256
257 // least square
258 // dpsi[0] = ((primin_array(i + 1, j + 1, k, Alpha1) - primin_array(i - 1, j + 1, k, Alpha1))
259 // + (primin_array(i + 1, j - 1, k, Alpha1) - primin_array(i - 1, j - 1, k, Alpha1))
260 // + (primin_array(i + 1, j, k, Alpha1) - primin_array(i - 1, j, k, Alpha1)))
261 // / (6.0 * dx[0]);
262 // dpsi[1] = ((primin_array(i + 1, j + 1, k, Alpha1) - primin_array(i + 1, j - 1, k, Alpha1))
263 // + (primin_array(i, j + 1, k, Alpha1) - primin_array(i, j - 1, k, Alpha1))
264 // + (primin_array(i - 1, j + 1, k, Alpha1) - primin_array(i - 1, j - 1, k, Alpha1)))
265 // / (6.0 * dx[1]);
266
267 // 计算法向量的模长
268 amrex::Real norm_factor = 0.0;
269 for (int dim = 0; dim < BL_SPACEDIM; ++dim) {
270 norm_factor += dpsi[dim] * dpsi[dim];
271 }
272 norm_factor = std::sqrt(norm_factor);
273
274 // 归一化法向量
275 amrex::GpuArray<amrex::Real, 3> n_l = { 0.0 };
276 if (norm_factor > 1e-8) { // 防止除以零
277 for (int dim = 0; dim < BL_SPACEDIM; ++dim) {
278 n_l[dim] = -dpsi[dim] / norm_factor; // 法向量指向alpha减小的方向
279 }
280 }
281
282 return n_l;
283}

322 // 开关函数：仅在界面区激活

324 Real S_switch = alpha_avg * (1.0 - alpha_avg); // ∈ [0,1]

364 // 计算法向速度在下侧的变化量
365 Real delta_v_down = prim_array(i, j, k, norm_velocity_direction)
366 - prim_array(i - offset_norm.x, j - offset_norm.y, k - offset_norm.z, norm_velocity_direction);
367 // 返回上下两侧变化量的绝对值的最小值

369 };

373 Real alpha_i = cal_numerator(i - offset_along.x, j - offset_along.y, k - offset_along.z) / (cl + std::max(ul, vl)); // 前一点的系数

377#endif

380 Real sqrt_rl = std::sqrt(rl);
381 Real sqrt_rr = std::sqrt(rr);
382
383 // 计算加权平均速度
384 Real VnBar = (Vn[L] * sqrt_rl + Vn[R] * sqrt_rr) / (sqrt_rl + sqrt_rr);
385 // Real VnBar = (Vn[L] + Vn[R]) / 2;
386

388 Real cBar = sqrt((cl * cl * sqrt_rl + cr * cr * sqrt_rr) / (sqrt_rl + sqrt_rr)
389 + 0.5 * (sqrt_rl * sqrt_rr) / ((sqrt_rl + sqrt_rr) * (sqrt_rl + sqrt_rr)) * (Vn[R] - Vn[L]) * (Vn[R] - Vn[L]));
390 // Real cBar = (cl + cr) / 2;
391
392 // 计算波速估计
393 Real Ma_local = std::max(std::abs(Vn[L] / cl), std::abs(Vn[R] / cr)), Ma_limit = 0.1;
394 Real phi = sin(std::min(1., Ma_local / Ma_limit) * M_PI_2);
395 Real SL = std::min(Vn[L] - cl, VnBar - cBar); // 左行波速
396 Real SR = std::max(Vn[R] + cr, VnBar + cBar); // 右行波速
397 // HLL 通量计算
398 // if (SL >= 0) {
399 // VarArray res = prim_to_flux(stateL, pp, norm);
400 // res[Alpha1] = 0;
401 // return res;
402 // } else if (SR <= 0) {
403 // VarArray res = prim_to_flux(stateR, pp, norm);
404 // res[Alpha1] = 0;
405 // return res;
406 // } else {
407 // VarArray UL = prim_to_cons(stateL, pp);
408 // VarArray UR = prim_to_cons(stateR, pp);
409 // VarArray FL = prim_to_flux(stateL, pp, norm);
410 // VarArray FR = prim_to_flux(stateR, pp, norm);
411
412 // VarArray res;

415 // }

420 amrex::Real Sstar = ((pr - pl) + (rl * Vn[L] * (SL - Vn[L]) - rr * Vn[R] * (SR - Vn[R]))) / (rl * (SL - Vn[L]) - rr * (SR - Vn[R]));
421
422 VarArray res;
423 // HLLC 通量计算
424 if (SL >= 0) {
425 res = prim_to_flux(stateL, pp, norm);
426 res[Alpha1] = 0;
427 } else if (Sstar > 0) {
428 VarArray UL = prim_to_cons(stateL, pp);
429 VarArray FL = prim_to_flux(stateL, pp, norm);
430 amrex::Real Usf = (SL - Vn[L]) / (SL - Sstar);
431 VarArray Ustar = { Usf * ar1l,
432 Usf * ar2l,
433 AMREX_D_DECL(Usf * (ul + norm[0] * (Sstar - Vn[L])) * rl,
434 Usf * (vl + norm[1] * (Sstar - Vn[L])) * rl,
435 Usf * (wl + norm[2] * (Sstar - Vn[L])) * rl),
436 Usf * (UL[Pressure] + (Sstar - Vn[L]) * (rl * Sstar + pl / (SL - Vn[L]))),
437 alpha1l };
438 for (int ivar = 0; ivar < ncomp - 1; ivar++) {
439 res[ivar] = FL[ivar] + SL * (Ustar[ivar] - UL[ivar]);
440 }
441#if BL_SPACEDIM == 2
