14// 输入：三个相邻点的phi值phi_im1, phi_i, phi_ip1，位置x，参数beta
15// 输出：THINC重构后的phi值

17amrex::Real phi_THINC(amrex::Real phi_im1, amrex::Real phi_i, amrex::Real phi_ip1, amrex::Real x, amrex::Real beta = 1.8)
18{
19 auto sgn = [](amrex::Real val) -> amrex::Real {
20 return val > 0 ? 1.0 : (val < 0 ? -1.0 : 0.0);
21 };
22
23 amrex::Real min_phi = std::min(phi_im1, phi_ip1); // 最小值
24 amrex::Real max_phi = std::max(phi_im1, phi_ip1); // 最大值
25 amrex::Real phi_bar = (max_phi + min_phi) / 2; // 平均值
26 amrex::Real phi_delta = (max_phi - min_phi) / 2; // 半差值
27 amrex::Real theta = sgn(phi_ip1 - phi_im1); // 符号
28
29 if ((phi_im1 - phi_i) * (phi_i - phi_ip1) > 0) { //(phi_i > min_phi && phi_i < max_phi) {
30 amrex::Real di = atanh(1.0 - 1e-8 - (2.0 - 2e-8) * phi_i) / beta; // 计算界面位置
31 // amrex::Real di = -(1.0 / (2.0 * beta)) * std::log((std::exp(2.0 * beta * phi_i) - 1.0) / (std::exp(2.0 * beta) - std::exp(2.0 * beta * phi_i)));
32 return 0.5 + (0.5 - 1e-8) * theta * tanh(beta * (x - theta * di)); // THINC重构
33
34 // amrex::Real di = atanh((phi_bar - phi_i) / phi_delta) / beta;
35 // return phi_bar + phi_delta * theta * tanh(beta * (x - theta * di));
36 } else {
37 return phi_i; // 不满足条件时直接返回中心值
38 }
39}

42amrex::Real phi_THINC3(amrex::Real phi_im1, amrex::Real phi_i, amrex::Real phi_ip1, amrex::Real x, amrex::Real beta = 1.8)
43{
44 auto sgn = [](amrex::Real val) -> amrex::Real {
45 return val > 0 ? 1.0 : (val < 0 ? -1.0 : 0.0);
46 };
47
48 amrex::Real min_phi = std::min(phi_im1, phi_ip1);
49 amrex::Real max_phi = std::max(phi_im1, phi_ip1);
50 amrex::Real phi_bar = (max_phi + min_phi) / 2;
51 amrex::Real phi_delta = (max_phi - min_phi) / 2;
52 amrex::Real theta = sgn(phi_ip1 - phi_im1);
53
54 if ((phi_im1 - phi_i) * (phi_i - phi_ip1) > 0) { // 单调性条件
55 // 使用新的重构公式
56 amrex::Real kappa = beta; // 将 beta 视为 kappa 参数
57 amrex::Real tanh_k4 = std::tanh(kappa / 4.0);
58 amrex::Real tanh_k2 = std::tanh(kappa / 2.0);
59
60 amrex::Real numerator = 2.0 * tanh_k4 * (phi_ip1 - phi_i) * (phi_i - phi_im1);
61 amrex::Real denominator = (phi_ip1 - phi_i) * (tanh_k2 - tanh_k4) + (phi_i - phi_im1) * (tanh_k2 + tanh_k4);
62
63 amrex::Real delta_u = numerator / denominator;
64
65 return phi_i + delta_u;
66 } else {
67 return phi_i; // 如果不满足单调性条件，直接返回 phi_i
68 }
69}

98amrex::GpuArray<VarArray, 2>
99primitive_reconstruction(int i, int j, int k,
100 amrex::Array4<amrex::Real const> const& primin_array,
101 int NUM_STATE, ProbParm const& pp,
102 amrex::GpuArray<amrex::Real, 3> norm,
103 amrex::Dim3 offset,
104 amrex::GpuArray<amrex::Real, AMREX_SPACEDIM> dx)
105{
106 // 初始化 qL 和 qR 用于存储左/右状态的物理量
107 amrex::GpuArray<amrex::GpuArray<amrex::Real, 5>, NVar> qL, qR;
108
109 // 遍历指定方向，计算每个位置的物理量
110 for (int ii = -3; ii <= 2; ii++) {
111 VarArray primj;
112 for (int ivar = 0; ivar < NUM_STATE; ivar++) {
113 primj[ivar] = primin_array(i + ii * offset.x, j + ii * offset.y, k + ii * offset.z, ivar);
114 }
115
116 int left_index = ii + 3;
117 if (left_index < 5) {
118 for (int ivar = 0; ivar < NUM_STATE; ivar++) {
119 qL[ivar][left_index] = primj[ivar];
120 }
121 }
122
123 int right_index = 5 - left_index;
124 if (right_index < 5) {
125 for (int ivar = 0; ivar < NUM_STATE; ivar++) {
126 qR[ivar][right_index] = primj[ivar];
127 }
128 }
129 }
130
131 // 对左、右状态的数据进行 WENO5 重构
132 VarArray primReconL, primReconR;
133 for (int ivar = 0; ivar < NUM_STATE; ivar++) {
134 primReconL[ivar] = weno_is(qL[ivar]);
135 primReconR[ivar] = weno_is(qR[ivar]);
136 }
137
138 VarArray primCellL, primCellR;
139
140 for (int ivar = 0; ivar < NUM_STATE; ivar++) {
141 primCellL[ivar] = primin_array(i - 1 * offset.x, j - 1 * offset.y, k - 1 * offset.z, ivar);
142 primCellR[ivar] = primin_array(i - 0 * offset.x, j - 0 * offset.y, k - 0 * offset.z, ivar);
143 }
144
145 // 将原始变量转换为守恒变量
146 VarArray consCellL = prim_to_cons(primCellL, pp);
147 VarArray consCellR = prim_to_cons(primCellR, pp);
148
149 VarArray consReconL = prim_to_cons(primReconL, pp);
150 VarArray consReconR = prim_to_cons(primReconR, pp);
151
152 // 应用保正限制器
153 const PositiveQuantityFunctions pqf;
154 apply_positive_limiter(consCellL, consReconL, pqf, pp, NUM_STATE);
155 apply_positive_limiter(consCellR, consReconR, pqf, pp, NUM_STATE);
156
157 // 将保正后的守恒变量转换回原始变量
158 VarArray primL = cons_to_prim(consReconL, pp);
159 VarArray primR = cons_to_prim(consReconR, pp);
160
161 // 返回拼接后的结果
162 return { primL, primR };
163}
